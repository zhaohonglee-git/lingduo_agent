## ItemSpec

Defines an item to be packed.

```typescript
interface ItemSpec {
  id: string;
  width: number;
  height: number;
  depth: number;
  weight?: number;
  quantity?: number;

  // Rotation constraints
  rotationType?: 'all' | 'horizontal' | 'fixed';
  allowedRotations?: RotationTypeValue[];

  // Stacking constraints
  maxStackWeight?: number;
  requiresFloor?: boolean;
  requiresSupport?: number;

  // Grouping
  groupId?: string;
  incompatibleWith?: string[];

  // Logistics
  deliveryOrder?: number;

  // Custom data
  metadata?: Record<string, unknown>;
}
```

## BinSpec

Defines a container/bin.

```typescript
interface BinSpec {
  id: string;
  type: 'box' | 'pallet' | 'container' | 'truck' | 'shelf' | 'custom';
  width: number;
  height: number;
  depth: number;
  maxWeight?: number;
  cost?: number;
  existingItems?: PlacedItem[];
  excludeZones?: AABB[];
}
```

## PackedBin

Result of packing items into a bin.

```typescript
interface PackedBin {
  bin: BinSpec;
  items: PlacedItem[];
  utilization: number;        // 0-1
  weight: number;
  centerOfGravity: Point3D;
}
```

## PlacedItem

An item with its placement information.

```typescript
interface PlacedItem {
  item: ItemSpec;
  position: Point3D;          // {x, y, z}
  rotation: RotationTypeValue;
  dimensions: Dimensions3D;   // After rotation
}
```

## PackOptions

Configuration options for packing.

```typescript
interface PackOptions {
  algorithm?: 'extreme-point' | 'layer-building' | 'wall-building' | 'eb-afit';
  preprocessor?: 'block-building' | 'none';
  enhancer?: 'genetic' | 'simulated-annealing' | 'tabu-search' | 'none';
  timeBudgetMs?: number;

  features?: {
    supportCheck?: boolean;
    weightBalance?: boolean;
    stackingLimit?: boolean;
  };

  constraints?: {
    respectDeliveryOrder?: boolean;
    keepGroupsTogether?: boolean;
    enforceIncompatibility?: boolean;
  };

  optimize?: {
    target?: 'space' | 'cost' | 'balanced';
  };

  binSelection?: 'first-fit' | 'best-fit' | 'spread';
}
```

## RotationType

Available rotation orientations.

```typescript
const RotationType = {
  RT_WHD: 0,  // Width × Height × Depth (original)
  RT_HWD: 1,  // Height × Width × Depth
  RT_HDW: 2,  // Height × Depth × Width
  RT_DHW: 3,  // Depth × Height × Width
  RT_DWH: 4,  // Depth × Width × Height
  RT_WDH: 5,  // Width × Depth × Height
}
```
